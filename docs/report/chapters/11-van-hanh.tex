\section{Audit vận hành}
Mọi tầng ghi trạng thái vận hành vào schema \code{audit} của PostgreSQL (danh sách bảng ở Phụ lục~\ref{app:audit}).
Bản lưu dài hạn của crawl audit, run batch và kết quả quality nằm trong lake (ADR-0008); bảng \code{audit.*} là bản vận hành
mà circuit breaker, Grafana và es-projector đọc.

\section{Giám sát}
\textbf{Grafana Alloy} là collector duy nhất: gom log container và log task Airflow vào Loki; đẩy metrics sang Prometheus gồm cAdvisor,
Kafka lag, Redis, Elasticsearch, StatsD của Airflow, các probe HTTP, và \textbf{truy vấn SQL trên bảng \code{audit.*}}
(\code{mp_crawl_*}, \code{mp_batch_*}, \code{mp_quality_*}, \code{mp_speed_*}\ldots). Grafana có 5 dashboard:
\code{lambda-overview}, \code{lambda-pipeline}, \code{lambda-resources}, \code{lambda-orchestration}, \code{lambda-logs}.

\section{Bảo trì, sao lưu và khôi phục (P8)}
DAG \code{marketplace_maintenance} (\code{@weekly}): \code{prune_raw} \ra\ \code{backup} \ra\ \code{storage_snapshot}.

\diagramslot{fig-4-7-dag-bao-tri}
  {DAG \texttt{marketplace\_maintenance} và nội dung backup}
  {Ba task chạy tuần tự; backup giữ batch lock và chép bốn nhóm: Gold (manifests, \texttt{current.json}, run được trỏ tới), Bronze và Silver, \texttt{pg\_dump} schema audit/cache, SHA-256 từng file.}
  {Hình 4.7 — \texttt{docs/system/04-pipeline.md} §4.10 (7 node)}
  {Bảo trì hằng tuần: dọn raw cũ, sao lưu, đo dung lượng.}

\begin{itemize}
  \item Backup chép Gold \textbf{trước}, sau đó Bronze và Silver; bị từ chối nếu \code{run_id} của pointer khác \code{audit.marketplace_cache_version.run_id}.
  \item Restore ghi \code{current.json} \textbf{cuối cùng}, sau đó chạy \code{validate --restored}.
  \item Không sao lưu: Kafka, Elasticsearch, Redis, checkpoint speed; những thứ này dựng lại được từ Bronze.
  \item Chưa có bảo trì Delta (\code{OPTIMIZE}, \code{VACUUM}) và chưa xoá run Gold cũ (rủi ro R4, Mục~\ref{sec:risks}).
\end{itemize}
