\section{Quản lý schema và hợp đồng dữ liệu}
\begin{itemize}
  \item \textbf{Hợp đồng có phiên bản trong tên}: \code{marketplace-record.v1}, \code{marketplace-observation.v1}, \code{marketplace-change.v1},
        \code{marketplace-crawl-audit.v1}, \code{marketplace-raw-artifact-metadata.v2}; topic kèm hậu tố \code{.v1}.
  \item \textbf{Schema theo nguồn}: \code{PAYLOAD_SCHEMA} trong normalizer. Khoá mới của sàn được báo trong \code{payload_drift} mà không chặn run.
  \item \textbf{Phiên bản luật} ghi trên từng dòng kết quả: \code{rule_version} (speed), \code{anomaly_rule_version}, \code{discount-rules.v1}, phiên bản luật freshness.
  \item \textbf{Phiên bản mã} ghi trên từng record: \code{adapter_version}, \code{extractor_version}.
\end{itemize}

\section{Bảo mật và quyền riêng tư}
\begin{itemize}
  \item \textbf{PII}: trường PII khai báo trong \code{SourceSpec.pii_row_fields}/\code{pii_context_fields} bị bỏ trước khi ghi records; tên trường đã bỏ được ghi vào \code{dropped_fields}.
        Raw vẫn giữ nguyên byte và bị xoá sau 30 ngày (\code{prune-raw}).
  \item \textbf{Bí mật}: biến môi trường theo stack (\code{env/<stack>.env}, không commit); Databricks lấy khoá S3 từ secret scope \code{ecomdata}.
  \item \textbf{Mạng}: cổng giám sát bind \code{127.0.0.1}; các service chỉ chạy trên máy cá nhân.
  \item \textbf{Thu thập có trách nhiệm}: robots.txt, throttle, user-agent tự khai, không vượt anti-bot (Chương~\ref{chap:ingest}).
\end{itemize}
