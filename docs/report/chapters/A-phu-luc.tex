\chapter{Tham số cấu hình chính}\label{app:config}
% =====================================================================
Cấu hình nạp bởi \code{settings/load_settings()} từ biến môi trường (\code{env/base.env}, \code{env/<stack>.env}).

\begin{xltabular}{\linewidth}{@{}L{70mm}Y@{}}
  \caption{Biến môi trường thường chỉnh}\label{tab:env}\\
  \toprule
  \textbf{Biến} & \textbf{Ý nghĩa / mặc định} \\
  \midrule
  \code{CRAWL_REQUEST_DELAY_SECONDS} & delay tối thiểu giữa hai request cùng host (lấy max với mức của nguồn) \\
  \code{CRAWL_<TIER>_CADENCE_MINUTES} & chu kỳ của tier (ACTIVE 60, HOT 120, NORMAL 240, COLD 720, SWEEP 1440) \\
  \code{MARKETPLACE_STREAM_TRIGGER} & trigger của speed (30 s) \\
  \code{MARKETPLACE_STREAM_CHECKPOINT_VERSION} & đổi để reset speed (đọc lại từ \code{earliest}, state rỗng) \\
  \code{MARKETPLACE_RAW_RETENTION_DAYS} & số ngày giữ Bronze/raw (30) \\
  \code{MARKETPLACE_COUNTER_MAX_GAP_SECONDS} & khoảng cách tối đa giữa hai lần thấy để delta counter hợp lệ (172.800) \\
  \code{MARKETPLACE_CRAWL_AUDIT_SOURCE} & \code{lake} (mặc định) hoặc \code{postgres} \\
  \code{DATA_LAKE_PROFILE} & một trong \code{local}, \code{minio}, \code{s3}, \code{r2}, \code{b2}; stack Docker dùng \code{minio}, Databricks dùng \code{s3} \\
  \code{SPARK_RUNTIME} & \code{local} hoặc \code{databricks} \\
  \code{DATA_LAKE_TABLE_FORMAT} & \code{delta} (mặc định) \\
  \bottomrule
\end{xltabular}
