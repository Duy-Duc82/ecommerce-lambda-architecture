\newcommand{\slotbox}[4]{%
  \begin{tikzpicture}
    \node[draw=muted!60,line width=0.4pt,
          minimum width=\linewidth-1pt,minimum height=#4,inner sep=0pt] {};
    \node[anchor=south east,font=\scriptsize\ttfamily,text=muted!70]
      at (current bounding box.south east) {figures/\detokenize\expandafter{\SlotFile}};
  \end{tikzpicture}}
\newcommand{\SlotFile}{}
\newcommand{\diagramslotH}[6]{% #1 chiều cao ô trống
  \begin{figure}[htbp]
    \centering
    \renewcommand{\SlotFile}{#2}%
    \IfFileExists{figures/#2.pdf}{\includegraphics[width=\linewidth,height=0.5\textheight,keepaspectratio]{#2.pdf}}{%
    \IfFileExists{figures/#2.png}{\includegraphics[width=\linewidth,height=0.5\textheight,keepaspectratio]{#2.png}}{%
    \IfFileExists{figures/#2.jpg}{\includegraphics[width=\linewidth,height=0.5\textheight,keepaspectratio]{#2.jpg}}{%
      \slotbox{#3}{#4}{#5}{#1}}}}
    \caption[#3]{#6}
    \label{fig:#2}
  \end{figure}}
